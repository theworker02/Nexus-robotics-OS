1//! Local robot gateway state machine. Networking transports plug in above this safety-preserving core.
2use nexus_core::SafetyState;
3use std::collections::VecDeque;
4#[derive(Clone, Copy, Debug, Eq, PartialEq)]
5pub enum GatewayState {
6    Discovered,
7    Connecting,
8    Authenticating,
9    Syncing,
10    Ready,
11    Degraded,
12    Offline,
13}
14#[derive(Clone, Debug, Eq, PartialEq)]
15pub enum GatewayError {
16    NotReady(GatewayState),
17    Offline,
18    SafetyStop,
19}
20impl std::fmt::Display for GatewayError {
21    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
22        write!(f, "{self:?}")
23    }
24}
25impl std::error::Error for GatewayError {}
26#[derive(Clone, Debug)]
27pub struct Gateway {
28    pub state: GatewayState,
29    pub safety: SafetyState,
30    telemetry_queue: VecDeque<String>,
31    pub local_emergency_control: bool,
32}
33impl Default for Gateway {
34    fn default() -> Self {
35        Self {
36            state: GatewayState::Discovered,
37            safety: SafetyState::Safe,
38            telemetry_queue: VecDeque::new(),
39            local_emergency_control: true,
40        }
41    }
42}
43impl Gateway {
44    pub fn advance(&mut self) {
45        self.state = match self.state {
46            GatewayState::Discovered => GatewayState::Connecting,
47            GatewayState::Connecting => GatewayState::Authenticating,
48            GatewayState::Authenticating => GatewayState::Syncing,
49            GatewayState::Syncing => GatewayState::Ready,
50            state => state,
51        };
52    }
53    pub fn accept_new_operation(&self) -> Result<(), GatewayError> {
54        if self.safety == SafetyState::EmergencyStop {
55            Err(GatewayError::SafetyStop)
56        } else if self.state == GatewayState::Offline {
57            Err(GatewayError::Offline)
58        } else if self.state == GatewayState::Ready {
59            Ok(())
60        } else {
61            Err(GatewayError::NotReady(self.state))
62        }
63    }
64    pub fn queue_telemetry(&mut self, event: impl Into<String>) {
65        self.telemetry_queue.push_back(event.into());
66    }
67    pub fn drain_telemetry(&mut self) -> Vec<String> {
68        self.telemetry_queue.drain(..).collect()
69    }
70    pub fn emergency_stop(&mut self) {
71        self.safety = SafetyState::EmergencyStop;
72        self.queue_telemetry("safety.triggered");
73    }
74    pub fn central_connection_lost(&mut self) {
75        self.state = GatewayState::Offline;
76        self.queue_telemetry("gateway.offline: unsafe new operations denied");
77    }
78}
79#[cfg(test)]
80mod tests {
81    use super::*;
82    #[test]
83    fn gateway_retains_safety_when_central_connection_drops() {
84        let mut gateway = Gateway::default();
85        for _ in 0..4 {
86            gateway.advance();
87        }
88        assert!(gateway.accept_new_operation().is_ok());
89        gateway.central_connection_lost();
90        assert_eq!(gateway.accept_new_operation(), Err(GatewayError::Offline));
91        assert!(!gateway.drain_telemetry().is_empty());
92        gateway.emergency_stop();
93        assert_eq!(gateway.safety, SafetyState::EmergencyStop);
94    }
95}
