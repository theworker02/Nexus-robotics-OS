9fn usage() {

12fn main() {
13    let args: Vec<String> = std::env::args()
14        .skip(1)
15        .filter(|value| value != "--no-ai")
16        .collect();
17    let command: Vec<&str> = args.iter().map(String::as_str).collect();
18    let mut runtime = if args.iter().any(|value| {
19        value == "nxr-2"
20            || value == "nori-a3"
21            || value == "warehouse-fetch"
22            || value == "fetch-object"
23            || value == "goal"
24            || value == "autonomy"
25            || value == "bench"
26            || value == "profile"
27            || value == "hardware"
28            || value == "emulate"
29            || value == "brain"
30            || value == "model"
31            || value == "upgrade"
32            || value == "simple-robot-test"
33    }) {
34        Runtime::nxr2()
35    } else {
36        Runtime::nxr1()
37    };
38    match command.as_slice() {
39        [] => usage(),
40        [cmd] if *cmd == "help" || *cmd == "--help" => usage(),
41        [cmd] if *cmd == "demo" || *cmd == "dev" => match runtime.run_warehouse_fetch_demo() { Ok(summary) => println!("Nexus Runtime READY\nSimulator READY\nNXR-2 CONNECTED\n{summary}\nReplay entries: {}", runtime.replays.values().map(|replay| replay.entries.len()).sum::<usize>()), Err(error) => fail(error), },
42        ["robot", "list", ..] => println!("{}\t{}\t{:?}\t{:.0}%", runtime.robot.identity, runtime.robot.capabilities.name, runtime.robot.health, runtime.robot.battery_percent),
43        ["robot", "inspect", ..] => { println!("{} ({})\nSafety: {}\nCapabilities:", runtime.robot.capabilities.name, runtime.robot.identity, runtime.robot.safety_state.label()); for capability in &runtime.robot.capabilities.capabilities { println!("- {}", capability.0); } },
44        ["robot", "diagnose", ..] => println!("joints: {} virtual servos\nsensors: {:?}\nnetwork: {}\nbattery: {:.0}%\nskills: {} enabled\nsafety: {}", runtime.virtual_bus.discover_servos().len(), runtime.robot.sensors, runtime.robot.network_connected, runtime.robot.battery_percent, runtime.skills.list().count(), runtime.robot.safety_state.label()),
45        ["profile", ..] => print_hardware_profile(&HardwareProfile::discovered_host()),
46        ["hardware", "validate", path] => match std::fs::read_to_string(path)
47            .map_err(|error| error.to_string())
48            .and_then(|content| HardwareProfile::from_manifest(&content)) {
49                Ok(profile) => {
50                    print_hardware_profile(&profile);
51                    let warnings = profile.validation_warnings();
52                    if warnings.is_empty() {
53                        println!("Hardware manifest: PASS");
54                    } else {
55                        println!("Hardware manifest: WARN");
56                        for warning in warnings { println!("- {warning}"); }
57                    }
58                }
59                Err(error) => eprintln!("hardware manifest invalid: {error}"),
60            },
61        ["emulate", "hardware", rest @ ..] => {
62            let mut profile = HardwareProfile::minimum_robot();
63            if let Some(value) = option(rest, "--ram").and_then(|value| value.parse().ok()) { profile.ram_mb = value; }
64            if let Some(value) = option(rest, "--cpu").and_then(|value| value.parse().ok()) { profile.cpu_cores = value; }
65            if let Some(value) = option(rest, "--camera").and_then(|value| value.parse().ok()) { profile.rgb_cameras = value; }
66            if let Some(value) = option(rest, "--motors").and_then(|value| value.parse().ok()) { profile.motors = value; }
67            profile.name = "Emulated hardware profile".into();
68            println!("HARDWARE EMULATION\nNo physical device or adapter was contacted.");
69            print_hardware_profile(&profile);
70        },
71        ["brain", "status", ..] => print_brain_plan(&runtime.hardware_profile, &runtime.brain_plan),
72        ["brain", "serve", ..] => eprintln!("Nexus Brain network serving is not implemented: authenticated pairing, mutual keys, and encrypted transport are required before a host can bind a network port."),
73        ["model", "recommend", ..] => print_model_recommendations(&runtime.hardware_profile),
74        ["upgrade", "advisor", rest @ ..] => {
75            let budget = option(rest, "--budget").and_then(|value| value.parse().ok());
76            let recommendations = upgrade_advisor(&runtime.hardware_profile, budget);
77            println!("Upgrade advisor: {}", runtime.brain_plan.class.label());
78            if recommendations.is_empty() { println!("No compatible recommendations within the supplied budget."); }
79            for recommendation in recommendations { println!("{}. {} (${})\n   {}", recommendation.priority, recommendation.title, recommendation.indicative_cost_usd, recommendation.benefit); }
80        },
81        ["skill", "list", ..] => for skill in runtime.skills.list() { let result = runtime.skills.compatibility(&skill.name, &runtime.robot.capabilities).expect("bundled skill exists"); println!("{}\t{}\t{}", skill.name, skill.version, if result.compatible { "SIMULATED" } else { "INCOMPATIBLE" }); },
82        ["skill", "test", skill] => match runtime.run_skill(skill, Some("blue_container")) { Ok(()) => println!("skill test PASS: {skill} on {}", runtime.robot.identity), Err(error) => fail(error) },
83        ["skill", "install", path] => match std::fs::read_to_string(path).ok().and_then(|yaml| runtime.skills.install_local_manifest(&yaml).ok()) { Some(name) => println!("installed local development skill: {name}"), None => eprintln!("unable to install: expected a readable skill.yaml with a name"), },
84        ["task", "run", task, rest @ ..] if *task == "fetch-object" => { if rest.iter().any(|value| *value == "--dry-run") { println!("DRY RUN PASS\nCapabilities: compatible\nSkills: inspect_object, walk_to, pick_up, place_object\nSafety: effective policy valid\nNo commands dispatched."); } else { match runtime.run_warehouse_fetch_demo() { Ok(summary) => println!("task completed: {summary}"), Err(error) => fail(error), } } },
85        ["task", "run", rest @ ..] => { let skill = option(rest, "--skill").unwrap_or("walk_to"); let target = option(rest, "--target"); if rest.iter().any(|value| *value == "--dry-run") { match runtime.skills.compatibility(skill, &runtime.robot.capabilities) { Ok(result) if result.compatible => println!("DRY RUN PASS: {skill}; safety and capability checks passed; no commands dispatched."), Ok(result) => println!("DRY RUN FAILED: missing {}", result.missing.join(", ")), Err(error) => fail(error), } } else { match runtime.run_skill(skill, target) { Ok(()) => println!("task completed: {skill}"), Err(error) => fail(error), } } },
86        ["sim", "robot", profile] if *profile == "nxr-2" => println!("NXR-2 CONNECTED\nSIMULATED mobile-manipulator profile\nVirtualBus: {} servos, {} camera streams", runtime.virtual_bus.discover_servos().len(), runtime.virtual_bus.camera_streams),
87        ["sim", "robot", profile] if *profile == "nori-a3" => println!("SIMULATED Nori-compatible profile CONNECTED\nCommunity Integration — not official hardware validation\nCapabilities: {}", nori_a3_manifest().records.len()),
88        ["sim", "scenario", scenario] if *scenario == "fetch_cube" => { let mut legacy = Runtime::nxr1(); match legacy.run_fetch_cube_demo() { Ok(result) => println!("scenario fetch_cube: {result}"), Err(error) => fail(error), } },
89        ["sim", "scenario", scenario] if *scenario == "warehouse-fetch" => match runtime.run_warehouse_fetch_demo() { Ok(result) => println!("scenario warehouse-fetch: {result}"), Err(error) => fail(error), },
90        ["sim", "scenario", scenario] if *scenario == "doorway-learning" => match runtime.run_unfamiliar_door_challenge() { Ok(result) => println!("scenario doorway-learning: {result}"), Err(error) => fail(error), },
91        ["sim", "scenario", scenario] if *scenario == "simple-robot-test" => match runtime.run_simple_robot_test() { Ok(result) => println!("scenario simple-robot-test: {result}"), Err(error) => fail(error), },
92        ["sim", "fault", fault] => { runtime.inject_fault(fault); println!("fault injected: {fault}; health={:?}", runtime.robot.health); },
93        ["goal", "plan", request @ ..] if !request.is_empty() => {
94            let request = request.join(" ");

96        },
97        ["goal", "run", request @ ..] if !request.is_empty() => {
98            let approved = request.iter().any(|value| *value == "--approve");
99            let request = request
100                .iter()
101                .filter(|value| **value != "--approve")
102                .copied()
103                .collect::<Vec<_>>()
104                .join(" ");
105            match runtime.run_goal(&request, approved) {
106                Ok(result) => println!("goal completed: {result}"),
107                Err(error) => fail(error),
108            }
109        },
110        ["autonomy", "envelope", ..] => print_envelope(&runtime.autonomy_envelope()),
111        ["autonomy", "profile", profile] => {
112            let profile = match *profile {
113                "manual" => IntelligenceProfile::Manual,
114                "assisted" => IntelligenceProfile::Assisted,
115                "supervised" => IntelligenceProfile::Supervised,
116                "autonomous" => IntelligenceProfile::Autonomous,
117                _ => {
118                    eprintln!("unknown autonomy profile; expected manual, assisted, supervised, or autonomous");
119                    return;
120                }
121            };
122            runtime.intelligence.policy = nexus_runtime::AutonomyPolicy::for_profile(profile);

124        },
125        ["bench", "simple-robot", ..] => match runtime.run_simple_robot_test() {

127            Err(error) => fail(error),
128        },
129        ["prove", "skill", skill, rest @ ..] => {
130            let trials = option(rest, "--trials").and_then(|value| value.parse().ok()).unwrap_or(100);
131            let seed = option(rest, "--seed").and_then(|value| value.parse().ok()).unwrap_or(483_208);
132            let report = ProvingGround::prove_skill(skill, trials, seed);
133            if let Some(path) = option(rest, "--output") {

135                    Ok(()) => println!("Proving Ground report written: {path}\nHighest earned: {}", report.highest_earned().label()),
136                    Err(error) => eprintln!("unable to write report: {error}"),
137                }
138            } else {
139                println!("{}", report.render_markdown());
140            }
141        },
142        ["prove", "all", rest @ ..] => {
143            let trials = option(rest, "--trials").and_then(|value| value.parse().ok()).unwrap_or(100);
144            for skill in ["fetch-object", "sense-hopping", "door-scan", "doorway-learning"] {
145                let report = ProvingGround::prove_skill(skill, trials, 483_208);
146                println!("{skill}: {} ({} / {} success; {} safe aborts)", report.highest_earned().label(), report.success_count(), report.trials.len(), report.safe_abort_count());
147            }
148        },
149        ["prove", "profile", profile] if *profile == "cheap-mobile-robot" => {
150            let profile = HardwareProfile::minimum_robot();
151            let plan = AdaptiveRuntimePlanner::plan(&profile);

153            println!("PROFILE EMULATION PASS\nProfile: {}\nRecommended intelligence: {}\nSenseHopping: {:?}\nStructureScan: {:?}\nWarnings: {}\nCompatibility: C1 Emulated (not HIL or hardware validated)", profile.name, plan.class.label(), plan.sense_hopping, plan.structure_scan, warnings.len());

158        ["structure", "diff", ..] => match runtime.structure_model.mutate_door("D-118", nexus_runtime::DoorState::Open) { Some(diff) => println!("StructureDiff: revision {} -> {}\n{:?}", diff.from_revision, diff.to_revision, diff.changes), None => eprintln!("structure diff failed"), },
159        ["learn", "create", ..] => { let session = doorway_learning_session(); println!("Learning session {}\nTask: {}\nBudget: {} attempts, {} seconds\nStage: {:?}", session.id, session.task, session.budget.max_attempts, session.budget.max_duration_s, session.stage); },
160        ["learn", "start", ..] => match runtime.run_unfamiliar_door_challenge() { Ok(_) => println!("Active Learning completed in simulation. Candidate proposal recorded in replay; no production skill was changed."), Err(error) => fail(error) },
161        ["learn", "status", ..] => println!("Learning mode: simulation-first\nProduction promotion: disabled\nPhysical learning: requires explicit hardware enablement and validation."),
162        ["compatibility", "inspect", profile] if *profile == "nori-a3" => { let manifest = nori_a3_manifest(); let compatible = runtime.skills.list().filter(|skill| runtime.skills.compatibility(&skill.name, &manifest.base).is_ok_and(|result| result.compatible)).count(); println!("NORI COMPATIBILITY SIMULATION\nRobot capabilities: {}\nCompatible built-in skills: {compatible}\nCommunity Integration; no vendor endorsement or physical-hardware claim.", manifest.records.len()); },
163        ["package", action, path] if *action == "validate" || *action == "inspect" => match std::fs::read_to_string(*path).ok().and_then(|content| PackageManifest::parse(&content).ok()) { Some(manifest) if *action == "validate" => println!("NRP valid: {} {} ({:?})", manifest.name, manifest.version, manifest.package_type), Some(manifest) => { let inspection = manifest.inspect(); println!("{}\nhash: {}\nsigned: {}\nproduction allowed: {}\n{}", manifest.name, inspection.content_hash, inspection.signed, inspection.production_allowed, inspection.warnings.join("\n")); }, None => eprintln!("invalid NRP manifest; required fields: name, version, type"), },
164        ["integration", "create", rest @ ..] => { let name = option(rest, "--name").unwrap_or("nexus-adapter"); let transport = option(rest, "--transport").unwrap_or("serial"); println!("{}", scaffold(name, transport, &["telemetry.read"])); },
165        ["telemetry", ..] => println!("battery={:.1}% pose=({:.2},{:.2}) network={} safety={} virtual={:?}", runtime.robot.battery_percent, runtime.robot.pose.x, runtime.robot.pose.y, runtime.robot.network_connected, runtime.robot.safety_state.label(), runtime.virtual_bus.telemetry()),
166        ["doctor", ..] => println!("runtime: OK\nsimulator: OK (NXR-1/NXR-2 deterministic)\nVirtualBus: OK\nNori adapter: community simulated profile available\nLeRobot adapter: dataset bridge available\nROS 2: planned optional adapter\nsecurity: unsigned NRP packages restricted to development"),
167        _ => usage(),
168    }

170fn option<'a>(args: &'a [&str], flag: &str) -> Option<&'a str> {
171    args.iter()
172        .position(|value| *value == flag)
173        .and_then(|index| args.get(index + 1))
174        .copied()
175}
176fn fail(error: impl std::fmt::Display) {
177    eprintln!("command failed: {error}");
178}

180fn print_goal_plan(plan: &nexus_runtime::GoalPlan) {
181    println!(
182        "Goal plan: {}\nState: {:?}\nRisk: {:?}\nExpected duration: {} seconds",
183        plan.goal.objective, plan.state, plan.risk, plan.expected_duration_s
184    );
185    for (index, step) in plan.steps.iter().enumerate() {
186        let permission = step
187            .permission
188            .map(|permission| format!("; permission={permission:?}"))
189            .unwrap_or_default();
190        let target = step
191            .target
192            .as_deref()
193            .map(|target| format!(" target={target}"))
194            .unwrap_or_default();
195        println!(
196            "{}. {}{}{}\n   {}",
197            index + 1,
198            step.skill,
199            target,
200            permission,
201            step.rationale
202        );
203    }
204    println!(
205        "Required capabilities: {}",
206        plan.required_capabilities.join(", ")
207    );
208}

270fn print_model_recommendations(profile: &HardwareProfile) {
271    let plan = AdaptiveRuntimePlanner::plan(profile);
272    println!(
273        "Model recommendations for {}\nSmall deterministic planner: supported locally",
274        profile.name
275    );
276    if plan.class >= nexus_runtime::IntelligenceClass::N2Adaptive {
277        println!(
278            "Compact perception model: eligible within MemCore model budget ({} MB)",
279            plan.memory.model_mb
280        );
281    } else {
282        println!("Compact perception model: not recommended locally; use an approved Brain host when available");
283    }
284    if plan.class >= nexus_runtime::IntelligenceClass::N3Intelligent {
285        println!("Large multimodal model: only after runtime-specific fit validation; current plan reserves {} MB", plan.memory.model_mb);
286    } else {
287        println!("Large multimodal model: insufficient local profile; remote placement requires a future paired Brain host");
288    }
289    println!("Safety and actuator control: LOCAL_REQUIRED; never delegated.");
