1//! Software watchdogs and dead-man motion expiry.
2use std::collections::BTreeMap;
3#[derive(Clone, Copy, Debug, Eq, PartialEq, Ord, PartialOrd)]
4pub enum WatchdogKind {
5    Command,
6    AdapterHeartbeat,
7    Telemetry,
8    Runtime,
9    Skill,
10}
11#[derive(Clone, Debug, Eq, PartialEq)]
12pub struct WatchdogAlert {
13    pub kind: WatchdogKind,
14    pub age_ms: u64,
15    pub limit_ms: u64,
16}
17#[derive(Default)]
18pub struct WatchdogSet {
19    last_seen_ms: BTreeMap<WatchdogKind, u64>,
20    limits_ms: BTreeMap<WatchdogKind, u64>,
21}
22impl WatchdogSet {
23    pub fn with_defaults() -> Self {
24        let limits_ms = [
25            (WatchdogKind::Command, 500),
26            (WatchdogKind::AdapterHeartbeat, 2_000),
27            (WatchdogKind::Telemetry, 5_000),
28            (WatchdogKind::Runtime, 2_000),
29            (WatchdogKind::Skill, 10_000),
30        ]
31        .into_iter()
32        .collect();
33        Self {
34            last_seen_ms: BTreeMap::new(),
35            limits_ms,
36        }
37    }
38    pub fn heartbeat(&mut self, kind: WatchdogKind, now_ms: u64) {
39        self.last_seen_ms.insert(kind, now_ms);
40    }
41    pub fn check(&self, now_ms: u64) -> Vec<WatchdogAlert> {
42        self.limits_ms
43            .iter()
44            .filter_map(|(kind, limit)| {
45                self.last_seen_ms
46                    .get(kind)
47                    .map(|seen| (kind, now_ms.saturating_sub(*seen), limit))
48            })
49            .filter(|(_, age, limit)| age > *limit)
50            .map(|(kind, age_ms, limit_ms)| WatchdogAlert {
51                kind: *kind,
52                age_ms,
53                limit_ms: *limit_ms,
54            })
55            .collect()
56    }
57}
58#[cfg(test)]
59mod tests {
60    use super::*;
61    #[test]
62    fn stale_motion_heartbeat_is_detected() {
63        let mut watchdogs = WatchdogSet::with_defaults();
64        watchdogs.heartbeat(WatchdogKind::Command, 0);
65        assert!(watchdogs
66            .check(501)
67            .iter()
68            .any(|alert| alert.kind == WatchdogKind::Command));
69    }
