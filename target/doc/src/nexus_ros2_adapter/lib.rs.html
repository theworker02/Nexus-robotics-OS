1//! ROS 2 capability discovery contract. Transport binding is intentionally separate.
2use nexus_core::{v2::*, Capability, CapabilityManifest, Health};
3use std::collections::BTreeMap;
4#[derive(Clone, Debug, Eq, PartialEq)]
5pub struct RosEndpoint {
6    pub name: String,
7    pub type_name: String,
8    pub kind: RosEndpointKind,
9}
10#[derive(Clone, Copy, Debug, Eq, PartialEq)]
11pub enum RosEndpointKind {
12    Topic,
13    Service,
14    Action,
15}
16pub fn map_endpoint(endpoint: &RosEndpoint) -> Option<&'static str> {
17    match endpoint.type_name.as_str() {
18        "sensor_msgs/msg/Image" => Some("vision.rgb"),
19        "sensor_msgs/msg/CameraInfo" => Some("vision.camera_calibration"),
20        "sensor_msgs/msg/Imu" => Some("sensing.imu"),
21        "sensor_msgs/msg/BatteryState" => Some("power.battery"),
22        "sensor_msgs/msg/PointCloud2" | "sensor_msgs/msg/LaserScan" => Some("sensing.lidar"),
23        "control_msgs/action/FollowJointTrajectory" => Some("manipulators.joint_trajectory"),
24        "geometry_msgs/msg/Twist" => Some("locomotion.control"),
25        _ => None,
26    }
27}
28pub struct Ros2Adapter {
29    endpoints: Vec<RosEndpoint>,
30    connected: bool,
31}
32impl Ros2Adapter {
33    pub fn new(endpoints: Vec<RosEndpoint>) -> Self {
34        Self {
35            endpoints,
36            connected: false,
37        }
38    }
39}
40impl RobotAdapter for Ros2Adapter {
41    fn name(&self) -> &str {
42        "ros2"
43    }
44    fn mode(&self) -> DeploymentMode {
45        DeploymentMode::Adapter
46    }
47    fn connect(&mut self) -> Result<(), AdapterError> {
48        self.connected = true;
49        Ok(())
50    }
51    fn disconnect(&mut self) -> Result<(), AdapterError> {
52        self.connected = false;
53        Ok(())
54    }
55    fn discover(&mut self) -> Result<CapabilityManifestV2, AdapterError> {
56        if !self.connected {
57            return Err(AdapterError::Unavailable(
58                "ROS 2 graph not connected".into(),
59            ));
60        }
61        let capabilities = self
62            .endpoints
63            .iter()
64            .filter_map(map_endpoint)
65            .map(Capability::new)
66            .collect();
67        let base = CapabilityManifest {
68            robot_id: "ros2-discovered".into(),
69            name: "ROS 2 discovered robot".into(),
70            architecture: "adapter".into(),
71            capabilities,
72        };
73        let mut manifest = CapabilityManifestV2::from_static(base, "ros2-adapter");
74        for record in manifest.records.values_mut() {
75            record.provenance.source_type = CapabilitySourceType::Discovered;
76        }
77        Ok(manifest)
78    }
79    fn health(&self) -> Health {
80        if self.connected {
81            Health::Healthy
82        } else {
83            Health::Offline
84        }
85    }
86    fn telemetry(&self) -> Result<BTreeMap<String, String>, AdapterError> {
87        Ok([
88            ("ros2.endpoints".into(), self.endpoints.len().to_string()),
89            ("ros2.connected".into(), self.connected.to_string()),
90        ]
91        .into_iter()
92        .collect())
93    }
94    fn stop(&mut self) -> Result<(), AdapterError> {
95        Ok(())
96    }
97}

101    #[test]
102    fn common_ros_interfaces_map_to_ncm() {
103        let mut adapter = Ros2Adapter::new(vec![
104            RosEndpoint {
105                name: "/cam".into(),
106                type_name: "sensor_msgs/msg/Image".into(),
107                kind: RosEndpointKind::Topic,
108            },
109            RosEndpoint {
110                name: "/base".into(),
111                type_name: "geometry_msgs/msg/Twist".into(),
112                kind: RosEndpointKind::Topic,
113            },
114        ]);
115        adapter.connect().unwrap();
116        let manifest = adapter.discover().unwrap();
117        assert!(manifest.supports("vision.rgb"));
118        assert!(manifest.supports("locomotion.control"));
119    }
